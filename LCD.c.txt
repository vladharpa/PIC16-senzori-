
#include "LCD.h"

void lcd_nibble(unsigned char date, unsigned char mod_rs) {
    RS = mod_rs; 
    D4 = (date & 0x01) ? 1 : 0;
    D5 = (date & 0x02) ? 1 : 0;
    D6 = (date & 0x04) ? 1 : 0;
    D7 = (date & 0x08) ? 1 : 0;
    EN = 1; __delay_ms(2); EN = 0; __delay_ms(2);
}

void lcd_cmd(unsigned char cmd) {
    lcd_nibble(cmd >> 4, 0);   
    lcd_nibble(cmd & 0x0F, 0); 
}

void lcd_data(char litera) {
    lcd_nibble(litera >> 4, 1);
    lcd_nibble(litera & 0x0F, 1);
}

void lcd_init(void) {
    __delay_ms(100); 
    lcd_nibble(0x03, 0); __delay_ms(5);
    lcd_nibble(0x03, 0); __delay_ms(5);
    lcd_nibble(0x03, 0); __delay_ms(5);
    lcd_nibble(0x02, 0); __delay_ms(5); 
    lcd_cmd(0x28); // Mod 4-bi?i
    lcd_cmd(0x0C); // Display ON
    lcd_cmd(0x01); // Clear
    __delay_ms(10);
}

void lcd_goto(unsigned char pos) { 
    lcd_cmd(pos); 
}

void lcd_puts(char *str) { 
    while(*str) lcd_data(*str++); 
}
